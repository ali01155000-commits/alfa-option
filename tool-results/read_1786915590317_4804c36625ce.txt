    54→# ============ Strategy Implementations ============
    55→class BaseStrategy:
    56→    def __init__(self, name: str):
    57→        self.name = name
    58→        self.price_history: deque = deque(maxlen=200)
    59→        self.last_signal: Optional[str] = None
    60→
    61→    def add_price(self, price: float, timestamp: float = None):
    62→        self.price_history.append({'price': price, 'time': timestamp or time.time()})
    63→
    64→    def get_signal(self) -> Optional[str]:
    65→        return None

    67→    def reset(self):
    68→        self.price_history.clear()
    69→        self.last_signal = None

    72→class MACrossStrategy(BaseStrategy):
    73→    def __init__(self, fast_period=5, slow_period=20):
    74→        super().__init__("ma_cross")
    75→        self.fast_period = fast_period
    76→        self.slow_period = slow_period
    77→
    78→    def get_signal(self) -> Optional[str]:
    79→        if len(self.price_history) < self.slow_period + 2:
    80→            return None
    81→        prices = [p['price'] for p in self.price_history]
    82→        fast_ma = sum(prices[-self.fast_period:]) / self.fast_period
    83→        slow_ma = sum(prices[-self.slow_period:]) / self.slow_period
    84→        prev_fast = sum(prices[-self.fast_period-1:-1]) / self.fast_period
    85→        prev_slow = sum(prices[-self.slow_period-1:-1]) / self.slow_period
    86→        if prev_fast <= prev_slow and fast_ma > slow_ma:
    87→            return 'call'
    88→        elif prev_fast >= prev_slow and fast_ma < slow_ma:
    89→            return 'put'
    90→        return None

    93→class RSIStrategy(BaseStrategy):
    94→    def __init__(self, period=14, overbought=70, oversold=30):
    95→        super().__init__("rsi")
    96→        self.period = period
    97→        self.overbought = overbought
    98→        self.oversold = oversold
    99→
   100→    def get_signal(self) -> Optional[str]:
   101→        if len(self.price_history) < self.period + 1:
   102→            return None
   103→        prices = [p['price'] for p in self.price_history]
   104→        deltas = [prices[i] - prices[i-1] for i in range(1, len(prices))]
   105→        recent = deltas[-(self.period):]
   106→        gains = [d if d > 0 else 0 for d in recent]
   107→        losses = [-d if d < 0 else 0 for d in recent]
   108→        avg_gain = sum(gains) / self.period
   109→        avg_loss = sum(losses) / self.period
   110→        if avg_loss == 0:
   111→            rsi = 100
   112→        else:
   113→            rs = avg_gain / avg_loss
   114→            rsi = 100 - (100 / (1 + rs))
   115→        if rsi < self.oversold:
   116→            return 'call'
   117→        elif rsi > self.overbought:
   118→            return 'put'
   119→        return None

   122→class MACDStrategy(BaseStrategy):
   123→    def __init__(self, fast=12, slow=26, signal=9):
   124→        super().__init__("macd")
   125→        self.fast = fast
   126→        self.slow = slow
   127→        self.signal = signal
   128→
   129→    def _ema(self, data: List[float], period: int) -> List[float]:
   130→        if len(data) < period:
   131→            return []
   132→        multiplier = 2 / (period + 1)
   133→        ema = [sum(data[:period]) / period]
   134→        for price in data[period:]:
   135→            ema.append((price - ema[-1]) * multiplier + ema[-1])
   136→        return ema
   137→
   138→    def get_signal(self) -> Optional[str]:
   139→        if len(self.price_history) < self.slow + self.signal + 2:
   140→            return None
   141→        prices = [p['price'] for p in self.price_history]
   142→        fast_ema = self._ema(prices, self.fast)
   143→        slow_ema = self._ema(prices, self.slow)
   144→        min_len = min(len(fast_ema), len(slow_ema))
   145→        if min_len < 3:
   146→            return None
   147→        macd_line = [f - s for f, s in zip(fast_ema[-min_len:], slow_ema[-min_len:])]
   148→        signal_line = self._ema(macd_line, self.signal)
   149→        if len(signal_line) < 2 or len(macd_line) < 2:
   150→            return None
   151→        curr_macd = macd_line[-1]
   152→        curr_signal = signal_line[-1]
   153→        prev_macd = macd_line[-2]
   154→        prev_signal = signal_line[-2]
   155→        if prev_macd <= prev_signal and curr_macd > curr_signal:
   156→            return 'call'
   157→        elif prev_macd >= prev_signal and curr_macd < curr_signal:
   158→            return 'put'
   159→        return None

   162→class ScalpingStrategy(BaseStrategy):
   163→    def __init__(self):
   164→        super().__init__("scalping")
   165→
   166→    def get_signal(self) -> Optional[str]:
   167→        if len(self.price_history) < 5:
   168→            return None
   169→        prices = [p['price'] for p in self.price_history]
   170→        last3 = prices[-3:]
   171→        if all(last3[i] > last3[i-1] for i in range(1, 3)):
   172→            if prices[-1] - prices[-3] < (prices[-3] * 0.0003):
   173→                return 'call'
   174→        if all(last3[i] < last3[i-1] for i in range(1, 3)):
   175→            if prices[-3] - prices[-1] < (prices[-3] * 0.0003):
   176→                return 'put'
   177→        if prices[-2] > prices[-3] and prices[-1] < prices[-2]:
   178→            return 'put'
   179→        if prices[-2] < prices[-3] and prices[-1] > prices[-2]:
   180→            return 'call'
   181→        return None

   184→class TrendFollowStrategy(BaseStrategy):
   185→    def __init__(self, ma_period=10, threshold=0.0002):
   186→        super().__init__("trend_follow")
   187→        self.ma_period = ma_period
   188→        self.threshold = threshold
   189→
   190→    def get_signal(self) -> Optional[str]:
   191→        if len(self.price_history) < self.ma_period + 1:
   192→            return None
   193→        prices = [p['price'] for p in self.price_history]
   194→        ma = sum(prices[-self.ma_period:]) / self.ma_period
   195→        current = prices[-1]
   196→        deviation = (current - ma) / ma
   197→        if deviation > self.threshold:
   198→            return 'call'
   199→        elif deviation < -self.threshold:
   200→            return 'put'
   201→        return None

   298→# ============ WebSocket Endpoint ============
   299→@app.websocket("/ws")
   300→async def websocket_endpoint(websocket: WebSocket):
   301→    await websocket.accept()
   302→    state.add_ws_client(websocket)
   303→    logger.info(f"WebSocket client connected. Total: {len(state.ws_clients)}")
   304→
   305→    try:
   306→        while True:
   307→            # Receive messages from client
   308→            data = await websocket.receive_text()
   309→            try:
   310→                msg = json.loads(data)
   311→                msg_type = msg.get("type")
   312→
   313→                if msg_type == "subscribe":
   314→                    logger.info(f"Client subscribed to price updates")
   315→                elif msg_type == "ping":
   316→                    await websocket.send_json({"type": "pong"})
   317→
   318→            except json.JSONDecodeError:
   319→                pass
   320→    except WebSocketDisconnect:
   321→        state.remove_ws_client(websocket)
   322→        logger.info(f"WebSocket client disconnected. Total: {len(state.ws_clients)}")
   323→    except Exception as e:
   324→        state.remove_ws_client(websocket)
   325→        logger.debug(f"WebSocket error: {e}")

   328→async def broadcast_to_clients(event_type: str, data: Dict):
   329→    """Broadcast event to all WebSocket clients"""
   330→    with state._ws_lock:
   331→        clients = list(state.ws_clients)
   332→
   333→    message = json.dumps({"type": event_type, **data})
   334→    for client in clients:
   335→        try:
   336→            await client.send_text(message)
   337→        except:
   338→            state.remove_ws_client(client)

   341→# ============ Price Polling Thread ============
   342→def price_polling_loop():
   343→    """Poll Expert Option for real-time prices"""
   344→    logger.info("Price polling thread started")
   345→
   346→    SIM_PRICES = {
   347→        "EUR/USD": 1.0850, "GBP/USD": 1.2650, "USD/JPY": 149.50,
   348→        "AUD/USD": 0.6520, "USD/CAD": 1.3650, "EUR/GBP": 0.8580,
   349→        "EUR/JPY": 162.25, "GBP/JPY": 189.10, "USD/CHF": 0.8820,
   350→        "NZD/USD": 0.5980, "BTC/USD": 43500.0, "ETH/USD": 2350.0,
   351→    }
   352→
   353→    import random
   354→    tick = 0
   355→
   356→    while True:
   357→        tick += 1
   358→
   359→        if state.connected and state.api and EO_AVAILABLE:
   360→            try:
   361→                # Try to get real prices from Expert Option global_values
   362→                if hasattr(global_value, 'subscriptions') and global_value.subscriptions:
   363→                    for asset_id, sub_data in global_value.subscriptions.items():
   364→                        pair_name = ASSET_MAP_REVERSE.get(asset_id, f"ID:{asset_id}")
   365→                        if isinstance(sub_data, dict) and 'price' in sub_data:
   366→                            price = float(sub_data['price'])
   367→                            state.price_cache[pair_name] = price
   368→                        elif isinstance(sub_data, (int, float)):
   369→                            state.price_cache[pair_name] = float(sub_data)
   370→            except Exception as e:
   371→                logger.debug(f"Price polling error: {e}")
   372→
   373→            # Balance update every 10 seconds
   374→            if tick % 10 == 0:
   375→                try:
   376→                    profile = state.api.Profile()
   377→                    if isinstance(profile, dict):
   378→                        new_balance = float(profile.get('balance', state.balance))
   379→                        if new_balance != state.balance:
   380→                            state.balance = new_balance
   381→                except:
   382→                    pass
   383→
   384→        # Simulate prices if no real data
   385→        if not state.connected or not state.price_cache:
   386→            for pair, base in SIM_PRICES.items():
   387→                change = random.gauss(0, base * 0.00005)
   388→                SIM_PRICES[pair] = round(base + change, 5 if base < 10 else 2 if base > 100 else 4)
   389→                state.price_cache[pair] = SIM_PRICES[pair]
   390→
   391→        time.sleep(1)

   394→# ============ Auto-Trading Engine ============
   395→def auto_trading_loop():
   396→    logger.info("Auto-trading thread started")
   397→    state.init_strategy(state.auto_config.get("strategy", "scalping"))
   398→
   399→    while True:
   400→        if not state.auto_trading or not state.connected:
   401→            time.sleep(2)
   402→            continue
   403→
   404→        state.reset_day()
   405→
   406→        with state._lock:
   407→            config = state.auto_config.copy()
   408→            daily_pnl = state.daily_pnl
   409→            open_count = len(state.open_trades)
   410→
   411→        if daily_pnl <= -config["maxDailyLoss"]:
   412→            logger.warning(f"⛔ Max daily loss: ${daily_pnl:.2f} / -${config['maxDailyLoss']}")
   413→            time.sleep(30)
   414→            continue
   415→
   416→        if daily_pnl >= config["maxDailyProfit"]:
   417→            logger.info(f"🎯 Max daily profit: ${daily_pnl:.2f} / ${config['maxDailyProfit']}")
   418→            time.sleep(30)
   419→            continue
   420→
   421→        if open_count >= config["maxConcurrentTrades"]:
   422→            time.sleep(5)
   423→            continue
   424→
   425→        pair = config["pair"]
   426→        current_price = state.price_cache.get(pair)
   427→        if not current_price:
   428→            time.sleep(2)
   429→            continue
   430→
   431→        if state.active_strategy:
   432→            state.active_strategy.add_price(current_price)
   433→            signal = state.active_strategy.get_signal()
   434→
   435→            if signal:
   436→                direction = signal
   437→                asset_id = ASSET_MAP.get(pair, 240)
   438→                is_demo = 1 if state.is_demo else 0
   439→                expiry_seconds = config["expiryMinutes"] * 60
   440→
   441→                logger.info(f"📊 Signal: {signal.upper()} on {pair} @ {current_price}")
   442→
   443→                try:
   444→                    if EO_AVAILABLE and state.api:
   445→                        result = state.api.Buy(
   446→                            amount=config["amount"],
   447→                            type=direction,
   448→                            assetid=asset_id,
   449→                            exptime=expiry_seconds,
   450→                            isdemo=is_demo,
   451→                            strike_time=int(time.time())
   452→                        )
   453→                        logger.info(f"✅ Trade: {direction.upper()} ${config['amount']} on {pair}")
   454→                    else:
   455→                        result = "simulated"
   456→                        logger.info(f"📝 Sim trade: {direction.upper()} ${config['amount']} on {pair}")
   457→
   458→                    trade_record = {
   459→                        "id": f"auto_{int(time.time())}_{direction}",
   460→                        "pair": pair,
   461→                        "direction": "buy" if direction == "call" else "sell",
   462→                        "callPut": direction,
   463→                        "amount": config["amount"],
   464→                        "assetId": asset_id,
   465→                        "expiryMinutes": config["expiryMinutes"],
   466→                        "entryPrice": current_price,
   467→                        "strategy": config["strategy"],
   468→                        "timestamp": int(time.time() * 1000),
   469→                        "isDemo": state.is_demo,
   470→                        "result": str(result)[:300] if result else None,
   471→                    }
   472→
   473→                    with state._lock:
   474→                        state.open_trades.append(trade_record)
   475→
   476→                    state.active_strategy.last_signal = signal
   477→
   478→                except Exception as e:
   479→                    logger.error(f"❌ Auto-trade error: {e}")
   480→
   481→        wait_time = min(config["expiryMinutes"] * 60 + 3, 30)
   482→        time.sleep(wait_time)
   483→

   485→# ============ Trade Expiry Checker ============
   486→def trade_expiry_loop():
   487→    logger.info("Trade expiry checker started")
   488→
   489→    while True:
   490→        if not state.open_trades:
   491→            time.sleep(5)
   492→            continue
   493→
   494→        now = time.time() * 1000
   495→        expired_trades = []
   496→
   497→        with state._lock:
   498→            for trade in state.open_trades[:]:
   499→                expiry_ms = trade.get("expiryMinutes", 1) * 60 * 1000
   500→                trade_age = now - trade.get("timestamp", now)
   501→                if trade_age >= expiry_ms:
   502→                    expired_trades.append(trade)
   503→                    state.open_trades.remove(trade)
   504→
   505→        for trade in expired_trades:
   506→            entry_price = trade.get("entryPrice", 0)
   507→            pair = trade.get("pair", "")
   508→            current_price = state.price_cache.get(pair, entry_price)
   509→            direction = trade.get("callPut", trade.get("direction", "call"))
   510→            amount = trade.get("amount", 0)
   511→            payout_pct = 80
   512→
   513→            if direction == "call":
   514→                won = current_price > entry_price
   515→            else:
   516→                won = current_price < entry_price
   517→
   518→            pnl = amount * (payout_pct / 100) if won else -amount
   519→
   520→            closed_trade = {
   521→                **trade,
   522→                "exitPrice": current_price,
   523→                "pnl": pnl,
   524→                "won": won,
   525→                "payoutPercent": payout_pct,
   526→                "closeTimestamp": int(now),
   527→            }
   528→
   529→            with state._lock:
   530→                state.trade_history.append(closed_trade)
   531→                state.daily_pnl += pnl
   532→
   533→            emoji = "🟢" if won else "🔴"
   534→            logger.info(f"{emoji} Trade: {direction.upper()} ${amount} on {pair} - {'WON' if won else 'LOST'} ${abs(pnl):.2f}")
   535→
   536→        time.sleep(5)

   667→@app.post("/api/trade")
   668→def place_trade(req: TradeRequest):
   669→    if not state.connected:
   670→        raise HTTPException(status_code=400, detail="Not connected to Expert Option")
   671→
   672→    if req.amount < 1 or req.amount > 100:
   673→        raise HTTPException(status_code=400, detail="Amount must be between $1 and $100")
   674→
   675→    asset_id = ASSET_MAP.get(req.pair, 240)
   676→    direction = "call" if req.direction in ("buy", "call") else "put"
   677→    is_demo = 1 if state.is_demo else 0
   678→    expiry_seconds = req.expiryMinutes * 60
   679→    current_price = state.price_cache.get(req.pair, 0)
   680→
   681→    try:
   682→        if EO_AVAILABLE and state.api:
   683→            result = state.api.Buy(
   684→                amount=req.amount,
   685→                type=direction,
   686→                assetid=asset_id,
   687→                exptime=expiry_seconds,
   688→                isdemo=is_demo,
   689→                strike_time=int(time.time())
   690→            )
   691→        else:
   692→            result = "simulated"
   693→
   694→        trade_record = {
   695→            "id": f"manual_{int(time.time())}_{direction}",
   696→            "pair": req.pair,
   697→            "direction": "buy" if direction == "call" else "sell",
   698→            "callPut": direction,
   699→            "amount": req.amount,
   700→            "assetId": asset_id,
   701→            "expiryMinutes": req.expiryMinutes,
   702→            "entryPrice": current_price,
   703→            "timestamp": int(time.time() * 1000),
   704→            "isDemo": state.is_demo,
   705→            "manual": True,
   706→            "result": str(result)[:300] if result else None,
   707→        }
   708→
   709→        with state._lock:
   710→            state.open_trades.append(trade_record)
   711→
   712→        logger.info(f"✅ Manual trade: {direction.upper()} ${req.amount} on {req.pair}")
   713→
   714→        return {"success": True, "trade": trade_record}
   715→
   716→    except Exception as e:
   717→        logger.error(f"❌ Trade error: {e}")
   718→        raise HTTPException(status_code=500, detail=str(e))
