    72→class MACrossStrategy(BaseStrategy):
    73→    def __init__(self, fast_period=5, slow_period=20):
    74→        super().__init__("ma_cross")
    75→        self.fast_period = fast_period
    76→        self.slow_period = slow_period
    77→
    78→    def get_signal(self) -> Optional[str]:
    79→        if len(self.price_history) < self.slow_period + 2:
    80→            return None
    81→        prices = [p['price'] for p in self.price_history]
    82→        fast_ma = sum(prices[-self.fast_period:]) / self.fast_period
    83→        slow_ma = sum(prices[-self.slow_period:]) / self.slow_period
    84→        prev_fast = sum(prices[-self.fast_period-1:-1]) / self.fast_period
    85→        prev_slow = sum(prices[-self.slow_period-1:-1]) / self.slow_period
    86→        if prev_fast <= prev_slow and fast_ma > slow_ma:
    87→            return 'call'
    88→        elif prev_fast >= prev_slow and fast_ma < slow_ma:
    89→            return 'put'
    90→        return None

    93→class RSIStrategy(BaseStrategy):
    94→    def __init__(self, period=14, overbought=70, oversold=30):
    95→        super().__init__("rsi")
    96→        self.period = period
    97→        self.overbought = overbought
    98→        self.oversold = oversold
    99→
   100→    def get_signal(self) -> Optional[str]:
   101→        if len(self.price_history) < self.period + 1:
   102→            return None
   103→        prices = [p['price'] for p in self.price_history]
   104→        deltas = [prices[i] - prices[i-1] for i in range(1, len(prices))]
   105→        recent = deltas[-(self.period):]
   106→        gains = [d if d > 0 else 0 for d in recent]
   107→        losses = [-d if d < 0 else 0 for d in recent]
   108→        avg_gain = sum(gains) / self.period
   109→        avg_loss = sum(losses) / self.period
   110→        if avg_loss == 0:
   111→            rsi = 100
   112→        else:
   113→            rs = avg_gain / avg_loss
   114→            rsi = 100 - (100 / (1 + rs))
   115→        if rsi < self.oversold:
   116→            return 'call'
   117→        elif rsi > self.overbought:
   118→            return 'put'
   119→        return None

   122→class MACDStrategy(BaseStrategy):
   123→    def __init__(self, fast=12, slow=26, signal=9):
   124→        super().__init__("macd")
   125→        self.fast = fast
   126→        self.slow = slow
   127→        self.signal = signal
   128→
   129→    def _ema(self, data: List[float], period: int) -> List[float]:
   130→        if len(data) < period:
   131→            return []
   132→        multiplier = 2 / (period + 1)
   133→        ema = [sum(data[:period]) / period]
   134→        for price in data[period:]:
   135→            ema.append((price - ema[-1]) * multiplier + ema[-1])
   136→        return ema
   137→
   138→    def get_signal(self) -> Optional[str]:
   139→        if len(self.price_history) < self.slow + self.signal + 2:
   140→            return None
   141→        prices = [p['price'] for p in self.price_history]
   142→        fast_ema = self._ema(prices, self.fast)
   143→        slow_ema = self._ema(prices, self.slow)
   144→        min_len = min(len(fast_ema), len(slow_ema))
   145→        if min_len < 3:
   146→            return None
   147→        macd_line = [f - s for f, s in zip(fast_ema[-min_len:], slow_ema[-min_len:])]
   148→        signal_line = self._ema(macd_line, self.signal)
   149→        if len(signal_line) < 2 or len(macd_line) < 2:
   150→            return None
   151→        curr_macd = macd_line[-1]
   152→        curr_signal = signal_line[-1]
   153→        prev_macd = macd_line[-2]
   154→        prev_signal = signal_line[-2]
   155→        if prev_macd <= prev_signal and curr_macd > curr_signal:
   156→            return 'call'
   157→        elif prev_macd >= prev_signal and curr_macd < curr_signal:
   158→            return 'put'
   159→        return None

   162→class ScalpingStrategy(BaseStrategy):
   163→    def __init__(self):
   164→        super().__init__("scalping")
   165→
   166→    def get_signal(self) -> Optional[str]:
   167→        if len(self.price_history) < 5:
   168→            return None
   169→        prices = [p['price'] for p in self.price_history]
   170→        last3 = prices[-3:]
   171→        if all(last3[i] > last3[i-1] for i in range(1, 3)):
   172→            if prices[-1] - prices[-3] < (prices[-3] * 0.0003):
   173→                return 'call'
   174→        if all(last3[i] < last3[i-1] for i in range(1, 3)):
   175→            if prices[-3] - prices[-1] < (prices[-3] * 0.0003):
   176→                return 'put'
   177→        if prices[-2] > prices[-3] and prices[-1] < prices[-2]:
   178→            return 'put'
   179→        if prices[-2] < prices[-3] and prices[-1] > prices[-2]:
   180→            return 'call'
   181→        return None

   184→class TrendFollowStrategy(BaseStrategy):
   185→    def __init__(self, ma_period=10, threshold=0.0002):
   186→        super().__init__("trend_follow")
   187→        self.ma_period = ma_period
   188→        self.threshold = threshold
   189→
   190→    def get_signal(self) -> Optional[str]:
   191→        if len(self.price_history) < self.ma_period + 1:
   192→            return None
   193→        prices = [p['price'] for p in self.price_history]
   194→        ma = sum(prices[-self.ma_period:]) / self.ma_period
   195→        current = prices[-1]
   196→        deviation = (current - ma) / ma
   197→        if deviation > self.threshold:
   198→            return 'call'
   199→        elif deviation < -self.threshold:
   200→            return 'put'
   201→        return None

   303→# ============ WebSocket Endpoint ============
   304→@app.websocket("/ws")
   305→async def websocket_endpoint(websocket: WebSocket):
   306→    await websocket.accept()
   307→    state.add_ws_client(websocket)
   308→    logger.info(f"WebSocket client connected. Total: {len(state.ws_clients)}")
   309→
   310→    try:
   311→        while True:
   312→            # Receive messages from client
   313→            data = await websocket.receive_text()
   314→            try:
   315→                msg = json.loads(data)
   316→                msg_type = msg.get("type")
   317→
   318→                if msg_type == "subscribe":
   319→                    logger.info(f"Client subscribed to price updates")
   320→                elif msg_type == "ping":
   321→                    await websocket.send_json({"type": "pong"})
   322→
   323→            except json.JSONDecodeError:
   324→                pass
   325→    except WebSocketDisconnect:
   326→        state.remove_ws_client(websocket)
   327→        logger.info(f"WebSocket client disconnected. Total: {len(state.ws_clients)}")
   328→    except Exception as e:
   329→        state.remove_ws_client(websocket)
   330→        logger.debug(f"WebSocket error: {e}")

   333→async def broadcast_to_clients(event_type: str, data: Dict):
   334→    """Broadcast event to all WebSocket clients"""
   335→    with state._ws_lock:
   336→        clients = list(state.ws_clients)
   337→
   338→    message = json.dumps({"type": event_type, **data})
   339→    for client in clients:
   340→        try:
   341→            await client.send_text(message)
   342→        except:
   343→            state.remove_ws_client(client)

   346→# ============ Price Polling Thread ============
   347→def price_polling_loop():
   348→    """Poll Expert Option for real-time prices"""
   349→    logger.info("Price polling thread started")
   350→
   351→    SIM_PRICES = {
   352→        "EUR/USD": 1.0850, "GBP/USD": 1.2650, "USD/JPY": 149.50,
   353→        "AUD/USD": 0.6520, "USD/CAD": 1.3650, "EUR/GBP": 0.8580,
   354→        "EUR/JPY": 162.25, "GBP/JPY": 189.10, "USD/CHF": 0.8820,
   355→        "NZD/USD": 0.5980, "BTC/USD": 43500.0, "ETH/USD": 2350.0,
   356→    }
   357→
   358→    import random
   359→    tick = 0
   360→
   361→    while True:
   362→        tick += 1
   363→
   364→        if state.connected and state.api and EO_AVAILABLE:
   365→            try:
   366→                # Try to get real prices from Expert Option global_values
   367→                if hasattr(global_value, 'subscriptions') and global_value.subscriptions:
   368→                    for asset_id, sub_data in global_value.subscriptions.items():
   369→                        pair_name = ASSET_MAP_REVERSE.get(asset_id, f"ID:{asset_id}")
   370→                        if isinstance(sub_data, dict) and 'price' in sub_data:
   371→                            price = float(sub_data['price'])
   372→                            state.price_cache[pair_name] = price
   373→                        elif isinstance(sub_data, (int, float)):
   374→                            state.price_cache[pair_name] = float(sub_data)
   375→            except Exception as e:
   376→                logger.debug(f"Price polling error: {e}")
   377→
   378→            # Balance update every 10 seconds
   379→            if tick % 10 == 0:
   380→                try:
   381→                    profile = state.api.Profile()
   382→                    if isinstance(profile, dict):
   383→                        new_balance = float(profile.get('balance', state.balance))
   384→                        if new_balance != state.balance:
   385→                            state.balance = new_balance
   386→                except:
   387→                    pass
   388→
   389→        # Simulate prices if no real data
   390→        if not state.connected or not state.price_cache:
   391→            for pair, base in SIM_PRICES.items():
   392→                change = random.gauss(0, base * 0.00005)
   393→                SIM_PRICES[pair] = round(base + change, 5 if base < 10 else 2 if base > 100 else 4)
   394→                state.price_cache[pair] = SIM_PRICES[pair]
   395→
   396→        time.sleep(1)

   399→# ============ Auto-Trading Engine ============
   400→def auto_trading_loop():
   401→    logger.info("Auto-trading thread started")
   402→    state.init_strategy(state.auto_config.get("strategy", "scalping"))
   403→
   404→    while True:
   405→        if not state.auto_trading or not state.connected:
   406→            time.sleep(2)
   407→            continue
   408→
   409→        state.reset_day()
   410→
   411→        with state._lock:
   412→            config = state.auto_config.copy()
   413→            daily_pnl = state.daily_pnl
   414→            open_count = len(state.open_trades)
   415→
   416→        if daily_pnl <= -config["maxDailyLoss"]:
   417→            logger.warning(f"⛔ Max daily loss: ${daily_pnl:.2f} / -${config['maxDailyLoss']}")
   418→            time.sleep(30)
   419→            continue
   420→
   421→        if daily_pnl >= config["maxDailyProfit"]:
   422→            logger.info(f"🎯 Max daily profit: ${daily_pnl:.2f} / ${config['maxDailyProfit']}")
   423→            time.sleep(30)
   424→            continue
   425→
   426→        if open_count >= config["maxConcurrentTrades"]:
   427→            time.sleep(5)
   428→            continue
   429→
   430→        pair = config["pair"]
   431→        current_price = state.price_cache.get(pair)
   432→        if not current_price:
   433→            time.sleep(2)
   434→            continue
   435→
   436→        if state.active_strategy:
   437→            state.active_strategy.add_price(current_price)
   438→            signal = state.active_strategy.get_signal()
   439→
   440→            if signal:
   441→                direction = signal
   442→                asset_id = ASSET_MAP.get(pair, 240)
   443→                is_demo = 1 if state.is_demo else 0
   444→                expiry_seconds = config["expiryMinutes"] * 60
   445→
   446→                logger.info(f"📊 Signal: {signal.upper()} on {pair} @ {current_price}")

   450→                        result = state.api.Buy(
   451→                            amount=config["amount"],
   452→                            type=direction,
   453→                            assetid=asset_id,
   454→                            exptime=expiry_seconds,

   456→                            strike_time=int(time.time())
   457→                        )
   458→                        logger.info(f"✅ Trade: {direction.upper()} ${config['amount']} on {pair}")
   459→                    else:
   460→                        result = "simulated"
   461→                        logger.info(f"📝 Sim trade: {direction.upper()} ${config['amount']} on {pair}")
   462→
   463→                    trade_record = {
   464→                        "id": f"auto_{int(time.time())}_{direction}",
   465→                        "pair": pair,
   466→                        "direction": "buy" if direction == "call" else "sell",
   467→                        "callPut": direction,
   468→                        "amount": config["amount"],
   469→                        "assetId": asset_id,
   470→                        "expiryMinutes": config["expiryMinutes"],
   471→                        "entryPrice": current_price,
   472→                        "strategy": config["strategy"],
   473→                        "timestamp": int(time.time() * 1000),
   474→                        "isDemo": state.is_demo,
   475→                        "result": str(result)[:300] if result else None,
   476→                    }
   477→
   478→                    with state._lock:
   479→                        state.open_trades.append(trade_record)
   480→
   481→                    state.active_strategy.last_signal = signal
   482→
   483→                except Exception as e:
   484→                    logger.error(f"❌ Auto-trade error: {e}")
   485→
   486→        wait_time = min(config["expiryMinutes"] * 60 + 3, 30)
   487→        time.sleep(wait_time)

   490→# ============ Trade Expiry Checker ============
   491→def trade_expiry_loop():
   492→    logger.info("Trade expiry checker started")
   493→
   494→    while True:
   495→        if not state.open_trades:
   496→            time.sleep(5)
   497→            continue
   498→
   499→        now = time.time() * 1000
   500→        expired_trades = []
   501→
   502→        with state._lock:
   503→            for trade in state.open_trades[:]:
   504→                expiry_ms = trade.get("expiryMinutes", 1) * 60 * 1000
   505→                trade_age = now - trade.get("timestamp", now)
   506→                if trade_age >= expiry_ms:
   507→                    expired_trades.append(trade)
   508→                    state.open_trades.remove(trade)
   509→
   510→        for trade in expired_trades:
   511→            entry_price = trade.get("entryPrice", 0)
   512→            pair = trade.get("pair", "")
   513→            current_price = state.price_cache.get(pair, entry_price)
   514→            direction = trade.get("callPut", trade.get("direction", "call"))
   515→            amount = trade.get("amount", 0)
   516→            payout_pct = 80
   517→
   518→            if direction == "call":
   519→                won = current_price > entry_price
   520→            else:
   521→                won = current_price < entry_price
   522→
   523→            pnl = amount * (payout_pct / 100) if won else -amount
   524→
   525→            closed_trade = {
   526→                **trade,
   527→                "exitPrice": current_price,
   528→                "pnl": pnl,
   529→                "won": won,
   530→                "payoutPercent": payout_pct,
   531→                "closeTimestamp": int(now),
   532→            }
   533→
   534→            with state._lock:
   535→                state.trade_history.append(closed_trade)
   536→                state.daily_pnl += pnl
   537→
   538→            emoji = "🟢" if won else "🔴"
   539→            logger.info(f"{emoji} Trade: {direction.upper()} ${amount} on {pair} - {'WON' if won else 'LOST'} ${abs(pnl):.2f}")
   540→
   541→        time.sleep(5)
