    72→    72→class MACrossStrategy(BaseStrategy):
    73→    73→    def __init__(self, fast_period=5, slow_period=20):
    74→    74→        super().__init__("ma_cross")
    75→    75→        self.fast_period = fast_period
    76→    76→        self.slow_period = slow_period
    77→    77→
    78→    78→    def get_signal(self) -> Optional[str]:
    79→    79→        if len(self.price_history) < self.slow_period + 2:
    80→    80→            return None
    81→    81→        prices = [p['price'] for p in self.price_history]
    82→    82→        fast_ma = sum(prices[-self.fast_period:]) / self.fast_period
    83→    83→        slow_ma = sum(prices[-self.slow_period:]) / self.slow_period
    84→    84→        prev_fast = sum(prices[-self.fast_period-1:-1]) / self.fast_period
    85→    85→        prev_slow = sum(prices[-self.slow_period-1:-1]) / self.slow_period
    86→    86→        if prev_fast <= prev_slow and fast_ma > slow_ma:
    87→    87→            return 'call'
    88→    88→        elif prev_fast >= prev_slow and fast_ma < slow_ma:
    89→    89→            return 'put'
    90→    90→        return None

    93→    93→class RSIStrategy(BaseStrategy):
    94→    94→    def __init__(self, period=14, overbought=70, oversold=30):
    95→    95→        super().__init__("rsi")
    96→    96→        self.period = period
    97→    97→        self.overbought = overbought
    98→    98→        self.oversold = oversold
    99→    99→
   100→   100→    def get_signal(self) -> Optional[str]:
   101→   101→        if len(self.price_history) < self.period + 1:
   102→   102→            return None
   103→   103→        prices = [p['price'] for p in self.price_history]
   104→   104→        deltas = [prices[i] - prices[i-1] for i in range(1, len(prices))]
   105→   105→        recent = deltas[-(self.period):]
   106→   106→        gains = [d if d > 0 else 0 for d in recent]
   107→   107→        losses = [-d if d < 0 else 0 for d in recent]
   108→   108→        avg_gain = sum(gains) / self.period
   109→   109→        avg_loss = sum(losses) / self.period
   110→   110→        if avg_loss == 0:
   111→   111→            rsi = 100
   112→   112→        else:
   113→   113→            rs = avg_gain / avg_loss
   114→   114→            rsi = 100 - (100 / (1 + rs))
   115→   115→        if rsi < self.oversold:
   116→   116→            return 'call'
   117→   117→        elif rsi > self.overbought:
   118→   118→            return 'put'
   119→   119→        return None

   122→   122→class MACDStrategy(BaseStrategy):
   123→   123→    def __init__(self, fast=12, slow=26, signal=9):
   124→   124→        super().__init__("macd")
   125→   125→        self.fast = fast
   126→   126→        self.slow = slow
   127→   127→        self.signal = signal
   128→   128→
   129→   129→    def _ema(self, data: List[float], period: int) -> List[float]:
   130→   130→        if len(data) < period:
   131→   131→            return []
   132→   132→        multiplier = 2 / (period + 1)
   133→   133→        ema = [sum(data[:period]) / period]
   134→   134→        for price in data[period:]:
   135→   135→            ema.append((price - ema[-1]) * multiplier + ema[-1])
   136→   136→        return ema
   137→   137→
   138→   138→    def get_signal(self) -> Optional[str]:
   139→   139→        if len(self.price_history) < self.slow + self.signal + 2:
   140→   140→            return None
   141→   141→        prices = [p['price'] for p in self.price_history]
   142→   142→        fast_ema = self._ema(prices, self.fast)
   143→   143→        slow_ema = self._ema(prices, self.slow)
   144→   144→        min_len = min(len(fast_ema), len(slow_ema))
   145→   145→        if min_len < 3:
   146→   146→            return None
   147→   147→        macd_line = [f - s for f, s in zip(fast_ema[-min_len:], slow_ema[-min_len:])]
   148→   148→        signal_line = self._ema(macd_line, self.signal)
   149→   149→        if len(signal_line) < 2 or len(macd_line) < 2:
   150→   150→            return None
   151→   151→        curr_macd = macd_line[-1]
   152→   152→        curr_signal = signal_line[-1]
   153→   153→        prev_macd = macd_line[-2]
   154→   154→        prev_signal = signal_line[-2]
   155→   155→        if prev_macd <= prev_signal and curr_macd > curr_signal:
   156→   156→            return 'call'
   157→   157→        elif prev_macd >= prev_signal and curr_macd < curr_signal:
   158→   158→            return 'put'
   159→   159→        return None

   162→   162→class ScalpingStrategy(BaseStrategy):
   163→   163→    def __init__(self):
   164→   164→        super().__init__("scalping")
   165→   165→
   166→   166→    def get_signal(self) -> Optional[str]:
   167→   167→        if len(self.price_history) < 5:
   168→   168→            return None
   169→   169→        prices = [p['price'] for p in self.price_history]
   170→   170→        last3 = prices[-3:]
   171→   171→        if all(last3[i] > last3[i-1] for i in range(1, 3)):
   172→   172→            if prices[-1] - prices[-3] < (prices[-3] * 0.0003):
   173→   173→                return 'call'
   174→   174→        if all(last3[i] < last3[i-1] for i in range(1, 3)):
   175→   175→            if prices[-3] - prices[-1] < (prices[-3] * 0.0003):
   176→   176→                return 'put'
   177→   177→        if prices[-2] > prices[-3] and prices[-1] < prices[-2]:
   178→   178→            return 'put'
   179→   179→        if prices[-2] < prices[-3] and prices[-1] > prices[-2]:
   180→   180→            return 'call'
   181→   181→        return None

   184→   184→class TrendFollowStrategy(BaseStrategy):
   185→   185→    def __init__(self, ma_period=10, threshold=0.0002):
   186→   186→        super().__init__("trend_follow")
   187→   187→        self.ma_period = ma_period
   188→   188→        self.threshold = threshold
   189→   189→
   190→   190→    def get_signal(self) -> Optional[str]:
   191→   191→        if len(self.price_history) < self.ma_period + 1:
   192→   192→            return None
   193→   193→        prices = [p['price'] for p in self.price_history]
   194→   194→        ma = sum(prices[-self.ma_period:]) / self.ma_period
   195→   195→        current = prices[-1]
   196→   196→        deviation = (current - ma) / ma
   197→   197→        if deviation > self.threshold:
   198→   198→            return 'call'
   199→   199→        elif deviation < -self.threshold:
   200→   200→            return 'put'
   201→   201→        return None

   310→   310→                msg = json.loads(data)

   312→   312→
   313→   313→                if msg_type == "subscribe":
   314→   314→                    logger.info(f"Client subscribed to price updates")

   317→   317→
   318→   318→            except json.JSONDecodeError:
   319→   319→                pass
   320→   320→    except WebSocketDisconnect:
   321→   321→        state.remove_ws_client(websocket)
   322→   322→        logger.info(f"WebSocket client disconnected. Total: {len(state.ws_clients)}")
   323→   323→    except Exception as e:

   341→   341→# ============ Price Polling Thread ============
   342→   342→def price_polling_loop():
   343→   343→    """Poll Expert Option for real-time prices"""
   344→   344→    logger.info("Price polling thread started")
   345→   345→
   346→   346→    SIM_PRICES = {
   347→   347→        "EUR/USD": 1.0850, "GBP/USD": 1.2650, "USD/JPY": 149.50,

   352→   352→

   398→   398→
   399→   399→    while True:

   401→   401→            time.sleep(2)
   402→   402→            continue
   403→   403→
   404→   404→        state.reset_day()
   405→   405→
   406→   406→        with state._lock:
   407→   407→            config = state.auto_config.copy()
   408→   408→            daily_pnl = state.daily_pnl
   409→   409→            open_count = len(state.open_trades)
   410→   410→
   411→   411→        if daily_pnl <= -config["maxDailyLoss"]:
   412→   412→            logger.warning(f"⛔ Max daily loss: ${daily_pnl:.2f} / -${config['maxDailyLoss']}")
   413→   413→            time.sleep(30)
   414→   414→            continue
   415→   415→
   416→   416→        if daily_pnl >= config["maxDailyProfit"]:

   418→   418→            time.sleep(30)
   419→   419→            continue
   420→   420→
   421→   421→        if open_count >= config["maxConcurrentTrades"]:
   422→   422→            time.sleep(5)
   423→   423→            continue
   424→   424→
   425→   425→        pair = config["pair"]
   426→   426→        current_price = state.price_cache.get(pair)
   427→   427→        if not current_price:
   428→   428→            time.sleep(2)
   429→   429→            continue
   430→   430→
   431→   431→        if state.active_strategy:
   432→   432→            state.active_strategy.add_price(current_price)
   433→   433→            signal = state.active_strategy.get_signal()
   434→   434→
   435→   435→            if signal:
   436→   436→                direction = signal
   437→   437→                asset_id = ASSET_MAP.get(pair, 240)
   438→   438→                is_demo = 1 if state.is_demo else 0
   439→   439→                expiry_seconds = config["expiryMinutes"] * 60
   440→   440→
   441→   441→                logger.info(f"📊 Signal: {signal.upper()} on {pair} @ {current_price}")
   442→   442→
   443→   443→                try:
   444→   444→                    if EO_AVAILABLE and state.api:
   445→   445→                        result = state.api.Buy(
   446→   446→                            amount=config["amount"],
   447→   447→                            type=direction,
   448→   448→                            assetid=asset_id,
   449→   449→                            exptime=expiry_seconds,
   450→   450→                            isdemo=is_demo,
   451→   451→                            strike_time=int(time.time())
   452→   452→                        )
   453→   453→                        logger.info(f"✅ Trade: {direction.upper()} ${config['amount']} on {pair}")
   454→   454→                    else:
   455→   455→                        result = "simulated"
   456→   456→                        logger.info(f"📝 Sim trade: {direction.upper()} ${config['amount']} on {pair}")
   457→   457→
   458→   458→                    trade_record = {
   459→   459→                        "id": f"auto_{int(time.time())}_{direction}",
   460→   460→                        "pair": pair,
   461→   461→                        "direction": "buy" if direction == "call" else "sell",
   462→   462→                        "callPut": direction,
   463→   463→                        "amount": config["amount"],
   464→   464→                        "assetId": asset_id,
   465→   465→                        "expiryMinutes": config["expiryMinutes"],
   466→   466→                        "entryPrice": current_price,
   467→   467→                        "strategy": config["strategy"],
   468→   468→                        "timestamp": int(time.time() * 1000),
   469→   469→                        "isDemo": state.is_demo,
   470→   470→                        "result": str(result)[:300] if result else None,
   471→   471→                    }
   472→   472→
   473→   473→                    with state._lock:
   474→   474→                        state.open_trades.append(trade_record)
   475→   475→
   476→   476→                    state.active_strategy.last_signal = signal
   477→   477→
   478→   478→                except Exception as e:
   479→   479→                    logger.error(f"❌ Auto-trade error: {e}")
   480→   480→
   481→   481→        wait_time = min(config["expiryMinutes"] * 60 + 3, 30)
   482→   482→        time.sleep(wait_time)

   485→   485→# ============ Trade Expiry Checker ============
   486→   486→def trade_expiry_loop():
   487→   487→    logger.info("Trade expiry checker started")
   488→   488→
   489→   489→    while True:
   490→   490→        if not state.open_trades:
   491→   491→            time.sleep(5)
   492→   492→            continue
   493→   493→
   494→   494→        now = time.time() * 1000
   495→   495→        expired_trades = []
   496→   496→
   497→   497→        with state._lock:
   498→   498→            for trade in state.open_trades[:]:
   499→   499→                expiry_ms = trade.get("expiryMinutes", 1) * 60 * 1000
   500→   500→                trade_age = now - trade.get("timestamp", now)
   501→   501→                if trade_age >= expiry_ms:
   502→   502→                    expired_trades.append(trade)
   503→   503→                    state.open_trades.remove(trade)
   504→   504→
   505→   505→        for trade in expired_trades:
   506→   506→            entry_price = trade.get("entryPrice", 0)
   507→   507→            pair = trade.get("pair", "")
   508→   508→            current_price = state.price_cache.get(pair, entry_price)
   509→   509→            direction = trade.get("callPut", trade.get("direction", "call"))
   510→   510→            amount = trade.get("amount", 0)
   511→   511→            payout_pct = 80
   512→   512→
   513→   513→            if direction == "call":
   514→   514→                won = current_price > entry_price
   515→   515→            else:
   516→   516→                won = current_price < entry_price
   517→   517→
   518→   518→            pnl = amount * (payout_pct / 100) if won else -amount
   519→   519→
   520→   520→            closed_trade = {
   521→   521→                **trade,
   522→   522→                "exitPrice": current_price,
   523→   523→                "pnl": pnl,
   524→   524→                "won": won,
   525→   525→                "payoutPercent": payout_pct,
   526→   526→                "closeTimestamp": int(now),
   527→   527→            }
   528→   528→
   529→   529→            with state._lock:
   530→   530→                state.trade_history.append(closed_trade)
   531→   531→                state.daily_pnl += pnl
   532→   532→
   533→   533→            emoji = "🟢" if won else "🔴"
   534→   534→            logger.info(f"{emoji} Trade: {direction.upper()} ${amount} on {pair} - {'WON' if won else 'LOST'} ${abs(pnl):.2f}")
   535→   535→
   536→   536→        time.sleep(5)
